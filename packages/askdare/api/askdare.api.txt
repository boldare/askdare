/** What the browser needs to run one bot's chat. Public: it never holds secrets (ADR-0008). */
interface ChatConfig {
  readonly protocolVersion: number;
  /** Path of the chat route, such as `/api/chat`. Same origin only (ADR-0011). */
  readonly basePath: string;
  readonly botId: string;
  readonly widget: ChatWidget;
  /** Whether the bot accepts a description of the page it is on. */
  readonly pageContext: boolean;
  /** Whether the bot lets visitors rate answers. Answers carry a handle when the route takes ratings. */

  readonly attachments?: ChatAttachments;

    readonly maxSeconds: number;

  readonly styles?: ChatStyles;
}
/**
 * Which proxy to trust when reading the client IP:
 * - `vercel`: Vercel overwrites `x-forwarded-for` and sets `x-real-ip`, so both are reliable.
 * - `heroku`: Heroku's router appends the client IP, so the last entry is reliable and anything to
 *   its left may be spoofed.
 * - a number `n`: there are `n` trusted proxies; take the n-th entry from the right.
 * - `auto` (default): `vercel` when `VERCEL` is set, otherwise one trusted proxy.
 *
 * Never expose `next start` without a proxy in front: the client then controls these headers.
 */
type TrustProxy = "auto" | "vercel" | "heroku" | number;
/** One rating, as `onFeedback` receives it (ADR-0012). */
interface ChatFeedbackReport {
  readonly bot: string;
  /** `null` when the visitor took their rating back. */
  readonly rating: "up" | "down" | null;
  /** What the visitor wrote, up to 500 characters. Treat it as personal data. */
  readonly comment?: string | undefined;
  /**
   * A pseudonym (HMAC) of the signed-in user's ID, or of the visitor's: the same value
   * `onUsage` reports, so ratings and costs line up.
   */
  readonly visitor: string;
  /** The signed-in user, verified (ADR-0009); absent for anonymous visitors. */
  readonly identity?: VisitorIdentity | undefined;
  /** The OpenAI conversation, for your own review of it. Never sent to the browser. */
  readonly conversationId: string;
  /** The OpenAI ID of the rated answer within the conversation. */
  readonly itemId: string;
}
/** Stable error codes sent to clients. The accompanying message never contains internals. */
declare const ERROR_CODES: readonly ["invalid_input", "unauthorized", "session_expired", "not_found", "rate_limited", "budget_exceeded", "bot_unavailable", "content_blocked", "timeout", "upstream_error", "internal_error"];
type ErrorCode = (typeof ERROR_CODES)[number];
declare const MESSAGE_END_REASONS: readonly ["completed", "max_steps", "max_output_tokens", "timeout", "content_filter", "aborted"];
type MessageEndReason = (typeof MESSAGE_END_REASONS)[number];
/** Tokens one turn used, summed over its model calls, as OpenAI reports them. */
interface ChatTokenUsage {
  readonly inputTokens: number;
  /** Input tokens read from the prompt cache; part of `inputTokens`. */
  readonly cachedInputTokens: number;
  /** Input tokens written to the prompt cache; part of `inputTokens`. */
  readonly cacheWriteTokens: number;
  readonly outputTokens: number;
  /** Output tokens spent on reasoning; part of `outputTokens`. */
  readonly reasoningTokens: number;
}
/** One function call of a turn. Its arguments and result are never reported. */
interface ChatToolCallReport {
  readonly name: string;
  /** Epoch milliseconds. */
  readonly startedAt: number;
  readonly durationMs: number;
  readonly outcome: "ok" | "error" | "timeout";
}
/**
 * One chat turn, as `onUsage` receives it (ADR-0012). It carries counters, timings and
 * pseudonyms only: never message text, tool arguments or results, IP addresses or raw IDs.
 */
interface ChatUsageReport {
  readonly bot: string;
  /** The model that answered, as OpenAI reports it; the configured model if none answered. */
  readonly model: string;
  /** A pseudonym (HMAC) of the conversation; undefined when none was created. */
  readonly conversation: string | undefined;
  /**
   * A pseudonym (HMAC) of the signed-in user's ID, or of the visitor's when nobody is signed in:
   * the same value OpenAI gets as `safety_identifier`.
   */
  readonly visitor: string;
  /** Epoch milliseconds. */
  readonly startedAt: number;
  readonly durationMs: number;
  /** `stopped` when the turn was aborted before it could answer. */
  readonly outcome: "completed" | "stopped" | "failed";
  /** Why the answer ended, when it did. */
  readonly finishReason?: MessageEndReason | undefined;
  /** The stable error code, when the turn failed. */
  readonly errorCode?: ErrorCode | undefined;
  /** What OpenAI reported for the model calls that finished; less than billed if one broke off. */
  readonly usage: ChatTokenUsage;
  /** Model calls in the turn: one, plus one after each round of tool calls. */
  readonly steps: number;
  readonly tools: readonly ChatToolCallReport[];
}
interface ChatHandlerOptions {
  readonly bots: readonly Bot[];
  /** Path of the catch-all route. Default `/api/chat` (route file `app/api/chat/[...path]/route.ts`). */
  readonly basePath?: string;
  /** Default `process.env.OPENAI_API_KEY`. */
  readonly openaiApiKey?: string;
  /** Token secret, at least 32 characters. Default `process.env.CHAT_SECRET`. */
  readonly secret?: string;
  /** Former secrets still accepted during rotation. Default: comma-separated `process.env.CHAT_SECRET_PREVIOUS`. */
  readonly previousSecrets?: readonly string[];
  /**
   * Secret the site's backend signs identity tokens with, for `askdare('identify')` (ADR-0009).
   * At least 32 random characters, different from `secret`. Default `process.env.IDENTITY_SECRET`;
   * without one, identity is off.
   */
  readonly identitySecret?: string;
  readonly limits?: ChatHandlerLimitsOptions;
  /**
   * Called for every server error (status 500 and up), for an error that ends a streamed answer,
   * and once a day when a bot-wide or the global budget runs out, for your monitoring, such as Sentry.
   * Defaults to one `console.error` line with a pointer to the troubleshooting guide. Never
   * receives message contents or API keys. May be async; nothing it throws reaches the request.
   */
  readonly onError?: (report: ChatErrorReport) => void | Promise<void>;
  /**
   * Called once per chat turn, after its answer has streamed, including turns that failed: the
   * tokens it used, how long it took, which tools it called. For cost tracking and tracing, e.g.
   * with `@boldare/askdare-otel`. Counters and pseudonyms only, never message text. May be async;
   * nothing it throws reaches the chat.
   */
  readonly onUsage?: (report: ChatUsageReport) => void | Promise<void>;
  /**
   * Called when a visitor rates an answer 👍 or 👎, with an optional comment, or takes a rating
   * back. Without it, answers can't be rated; a bot can opt out with `widget.feedback: false`.
   * askdare stores no ratings: keep them here. May be async; nothing it throws reaches the chat.
   */
  readonly onFeedback?: (report: ChatFeedbackReport) => void | Promise<void>;
  /**
   * Serves the state gallery: `/embed/<bot>?gallery=<state>&preset=<preset>&scheme=<scheme>`
   * renders fixture states without calling the model. For development and visual tests; leave
   * it off in production.
   */
  readonly gallery?: boolean;
}
/** One server error, as `onError` receives it. */
interface ChatErrorReport {
  /** The stable code the widget also receives, such as `bot_unavailable` or `upstream_error`. */
  readonly code: ErrorCode;
  /** What went wrong, for you: a missing key, a model the project can't use. Never shown to visitors. */
  readonly message: string;
  /** The underlying error, with any API key cut down to its prefix. */
  readonly cause?: unknown;
}
interface ChatHandlerLimitsOptions {
  /**
   * Counter storage for rate limits and budgets. Defaults to per-process memory, which is weak when
   * several instances serve traffic; use `@boldare/askdare-upstash` there.
   */
  readonly store?: RateLimitStore;
  /** How to read the client IP behind your proxy. Default `auto` (Vercel, otherwise one proxy). */
  readonly trustProxy?: TrustProxy;
  /** Input plus output tokens per UTC day across all bots. Unlimited by default. */
  readonly globalDailyTokens?: number;
}
type RouteHandler = (request: Request) => Promise<Response>;
interface ChatHandlers {
  readonly GET: RouteHandler;
  readonly POST: RouteHandler;
  readonly DELETE: RouteHandler;

/**
 * Creates the Next.js route handlers for askdare. Mount them in a catch-all route:
 *
 * ```ts
 * // app/api/chat/[...path]/route.ts
 * export const maxDuration = 300;
 * export const { GET, POST, DELETE } = createChatHandler({ bots });
 * ```
 *
 * Secrets are read on the first request, so importing the route (for example during
 * `next build`) never requires them.
 */
export declare function createChatHandler(options: ChatHandlerOptions): ChatHandlers;
interface ChatConfigOptions {
  /** The same bots as in `createChatHandler`. */
  readonly bots: readonly Bot[];
  /** The id of the bot to show. */
  readonly bot: string;
  /** Path of the chat route, the same as in `createChatHandler`. Default `/api/chat`. */
  readonly basePath?: string;
  /**
   * The page's language, for a bot with `locale: 'auto'`; default English. A bot with a fixed
   * locale keeps it.
   */
  readonly locale?: string;
}
/**
 * The public configuration of one bot for `<Root config={…}>` from `@boldare/askdare/react`. Call
 * it in a Server Component and pass the result down: it holds only what browsers may see.
 *
 * ```tsx
 * // app/help/page.tsx
 * const config = getChatConfig({ bots, bot: 'support' });
 * return <Chat config={config} />; // a client component with <Root> and <DefaultChat>
 * ```
 */
export declare function getChatConfig(options: ChatConfigOptions): ChatConfig;

  /** Text on the pill launcher; shortcut for `texts['launcher.label']`. */
  readonly launcherLabel?: string;
  /** Built-in look (`halo`, `aura`, `sage`, `care`) or a theme from `defineTheme`. Default `aura`. */
  readonly theme?: PresetName | ThemeDefinition;
  /** Default `auto`: follows the visitor's system setting. */
  readonly colorScheme?: ColorScheme;
}
/** Widget settings with defaults applied and the theme resolved. */
interface BotWidget {
  /** In the configured locale (English for `auto`); the visitor may see another language. */
  readonly title: string;
  readonly subtitle: string;
  readonly greeting: string;
  readonly placeholder: string;
  /** The configured BCP 47 tag, or `auto`. */
  readonly locale: string;
  readonly starters: readonly string[];
  /** Whether this bot lets visitors rate answers, when the handler has `onFeedback`. */
  readonly feedback: boolean;
  readonly avatar: string | undefined;
  readonly launcherLabel: string;
  /** Text overrides, including the title, greeting, placeholder and launcher label shortcuts. */
  readonly texts: Readonly<Partial<Record<TextKey, string>>>;
  readonly disclosure: false | {
    readonly privacyUrl: string | undefined;
  };
  readonly css: string;
  readonly theme: ResolvedTheme;
}
interface BotDefinition {
  /** URL-safe identifier used in the script tag (`widget.js?bot=<id>`) and routes. */
  readonly id: string;
  /** OpenAI model ID. Required on purpose: the library never assumes a model (ADR-0002). */
  readonly model: string;
  readonly instructions: string | ((context: BotContext) => string);
  readonly limits?: BotLimits;
  /** Server-side functions the model may call; see `defineTool`. Keep it under about 20. */
  readonly tools?: readonly Tool[];
  /**
   * Accept a description of the page from `askdare('configure', { context })` and pass it to the
   * model with the visitor's message. Any script on the page, or the visitor, can set it, so it
   * gets no more trust than what visitors type. Off by default.
   */
  readonly pageContext?: boolean;
  /** Documents the model searches with OpenAI file search and cites in its answers. */
  readonly knowledge?: KnowledgeConfig | undefined;
  /**
   * Lets visitors attach images and PDFs. Off when absent. Files go to OpenAI (purpose
   * `user_data`) and expire after 30 days, or when the visitor deletes the conversation.
   */
  readonly attachments?: AttachmentConfig | undefined;
  /**
   * Lets visitors dictate messages: speech is transcribed by OpenAI into the message field, where
   * they can check it before sending. Off when absent.
   */
  readonly voice?: VoiceConfig | undefined;
  readonly security?: BotSecurity;
  readonly widget?: WidgetConfig;
}
interface Bot {
  readonly id: string;
  readonly model: string;
  readonly instructions: (context: BotContext) => string;
  readonly limits: Required<BotLimits>;
  readonly tools: readonly Tool[];
  readonly pageContext: boolean;
  readonly knowledge: Required<KnowledgeConfig> | undefined;
  readonly attachments: Required<AttachmentConfig> | undefined;
  readonly voice: Required<VoiceConfig> | undefined;
  readonly security: {
    readonly allowedOrigins: readonly string[];
  };
  readonly widget: BotWidget;
}
/**
 * Declares a chatbot. Validates eagerly so misconfiguration fails at startup, not on the first
 * visitor's message.
 */
declare function defineBot(definition: BotDefinition): Bot;
/** Result of incrementing a fixed-window counter. */
interface CounterHit {
  /** Counter value after the increment. */
  readonly count: number;
  /** Milliseconds until the window resets. */
  readonly resetMs: number;
}
/**
 * Shared counter storage behind rate limits and budgets. The in-memory default works for a single
 * instance; multi-instance deployments (Vercel, several dynos) need a shared store such as
 * `@boldare/askdare-upstash`. Implementations must be atomic per key.
 */
interface RateLimitStore {
  /**
   * Adds `amount` to `key`, starting a `windowMs` window if none is open. `0` reads without
   * changing; negative amounts refund.
   */
  hit(key: string, windowMs: number, amount: number): Promise<CounterHit>;
  /** True for stores that are shared across server instances. */
  readonly shared: boolean;
}
