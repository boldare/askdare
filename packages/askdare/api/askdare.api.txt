/**
 * The challenge the chat shows before a new session starts. Public: it is sent to browsers, so it
 * holds the site key and never the secret.
 */
interface HumanChallenge {
  /** Which widget the chat renders. Cloudflare Turnstile is the only one in 0.5. */
  readonly provider: "turnstile";
  /** The widget's public site key. */
  readonly siteKey: string;
}
/** What a verifier gets with each token. */
interface HumanCheckContext {
  /** The bot the session is for. */
  readonly botId: string;
  /** Aborts when the check takes too long or the visitor goes away; pass it to `fetch`. */
  readonly signal: AbortSignal;
}
/**
 * Checks that a new session comes from a person (ADR-0004, ADR-0019). The chat renders
 * `challenge`, and `/session` passes the token it got to `verify`. Renewals of a live session
 * aren't checked again. Build one with `turnstile()`, or implement it to check the token some
 * other way.
 */
interface HumanVerifier {
  readonly challenge: HumanChallenge;
  /**
   * Resolves true when the token proves a person, false when it doesn't (invalid, expired or
   * already used): the visitor is asked to try again. Throw when the check itself can't run, such
   * as a missing secret or an outage: the session is refused and `onError` hears about it.
   */
  verify(token: string, context: HumanCheckContext): Promise<boolean>;
}

  readonly name?: string | undefined;
  readonly email?: string | undefined;
  /** Other string, number or boolean claims of the token, such as a plan or a role. */
  readonly claims: Readonly<Record<string, string | number | boolean>>;
}

interface ToolContext {

  readonly locale: string;

/**
 * A value as `JSON.parse(JSON.stringify(value))` returns it: `toJSON()` applied (a `Date` becomes
 * a string), functions, symbols and `undefined` fields gone.
 */
type Jsonified<T> = T extends {
  toJSON(): infer R;
} ? Jsonified<R> : T extends bigint | symbol | ((...args: never[]) => unknown) | undefined ? never : T extends readonly (infer Item)[] ? Jsonified<Item>[] : T extends object ? { [Key in keyof T as Jsonified<T[Key]> extends never ? never : Key]: Jsonified<T[Key]>; } : T;
interface ToolDefinition<Input extends z.ZodObject, Output = unknown, Ui = unknown, Name extends string = string> {
  /** Name the model calls, 1–64 letters, digits, `_` or `-`. */
  readonly name: Name;
  /** Tells the model when and how to use the tool. */
  readonly description: string;
  /**
   * Arguments schema. OpenAI strict mode requires every field: use `.nullable()` instead of
   * `.optional()`.
   */
  readonly input: Input;
  /** Runs on the server with validated arguments. The result is sent back to the model as JSON. */
  readonly execute: (input: z.infer<Input>, context: ToolContext) => Output | PromiseLike<Output>;
  /** Default 10 s. */
  readonly timeoutMs?: number;
  /**
   * Options the visitor can click under the answer, derived from a successful result (ADR-0015).
   * It gets the result as the model received it, parsed back from JSON (a `Date` is a string), so
   * the options come back the same after a reload. A call that failed defines none. At most 4 of
   * up to 80 characters are shown, plain text without links; the others are dropped. The visitor
   * sends a clicked option as their own message. Only the latest answer shows options; when
   * several calls of a turn define them, the last one wins.
   */
  readonly replies?: (output: Jsonified<Output>) => readonly string[] | undefined;
  /**
   * What the visitor may see of a successful result, for a component the site registers with
   * `toolRenderers` (ADR-0017). Only this reaches the browser, never the result itself, so pick
   * the fields a visitor may see. It gets the result as the model received it, parsed back from
   * JSON, and runs again on every reload, so keep it pure: the same result, the same payload.
   * Return plain JSON of at most 8 KB and 8 levels, or `undefined` or `null` for none; a payload
   * over a limit is dropped whole. An answer shows at most 4, in call order.
   */
  readonly ui?: (output: Jsonified<Output>) => Ui | null | undefined;
}
/**
 * A tool the bot can call. `Name` and `Ui` carry the tool's name and the type its `ui` returns,
 * so a page can type its renderers with `import type` (ADR-0017).
 */
interface Tool<Name extends string = string, Ui = unknown> {
  readonly name: Name;
  readonly description: string;
  readonly input: z.ZodObject;
  readonly execute: (input: unknown, context: ToolContext) => unknown;
  readonly timeoutMs: number;
  readonly replies?: ((output: unknown) => readonly string[] | undefined) | undefined;
  readonly ui?: ((output: unknown) => Ui | null | undefined) | undefined;
}
/**
 * Declares a server-side function the model may call. Arguments are validated with the schema
 * before `execute` runs; failures are reported to the model, never to the visitor.
 */
declare function defineTool<Input extends z.ZodObject, Output = unknown, Ui = never, const Name extends string = string>(definition: ToolDefinition<Input, Output, Ui, Name>): Tool<Name, Ui>;
/**
 * OpenAI moderation for a bot's messages (ADR-0019). Each checked message costs one call to the
 * moderation endpoint before the turn goes on (docs/costs.md).
 */
interface ModerationConfig {
  /**
   * OpenAI moderation model, for example `omni-moderation-latest`. Required: the library never
   * assumes a model.
   */
  readonly model: string;
  /**
   * What is checked: the visitor's message before it reaches the model, the finished answer
   * before the visitor sees it, or both. Checking output holds each answer back until it is
   * complete (docs/guardrails.md). Default `input`.
   */
  readonly check?: "input" | "output" | "both";
  /**
   * Moderation categories that block, such as `harassment` or `self-harm/intent`. Default: any
   * category the model flags.
   */
  readonly categories?: readonly string[];
  /** What the visitor sees when moderation blocks a message. Default: the `error.content_blocked` text. */
  readonly message?: string;
}
/** Who and what a guardrail hook is checking, besides the text itself. */
interface GuardContext {
  /** The bot's `id`. */
  readonly bot: string;
  /** Validated BCP 47 tag chosen by the widget. */
  readonly locale: string;
  /** The signed-in user, verified with IDENTITY_SECRET (ADR-0009); undefined when anonymous. */
  readonly identity?: VisitorIdentity | undefined;
}
/**
 * What an `onInput` or `onOutput` hook decides: let the text through, block the turn (with an
 * optional text for the visitor, at most 300 characters), or replace the text. Returning
 * `undefined` lets it through.
 */
type GuardVerdict = {
  readonly action: "allow";
} | {
  readonly action: "block";
  readonly message?: string;
} | {
  readonly action: "redact";
  readonly text: string;
};
/** An `onInput` or `onOutput` hook. Runs on the server; whatever it throws lets the text through. */
type GuardHook = (text: string, context: GuardContext) => GuardVerdict | undefined | Promise<GuardVerdict | undefined>;
/** The guardrail options of `defineBot` (ADR-0019). Every one is off when absent. */
interface GuardrailOptions {
  /** OpenAI moderation on input, output or both. Off when absent. */
  readonly moderation?: ModerationConfig | undefined;
  /**
   * Your own rule for each visitor message, before it reaches the model (and before moderation):
   * allow it, block the turn, or replace the text the model gets. Files and the page's
   * description are not passed to it.
   */
  readonly onInput?: GuardHook | undefined;
  /**
   * Your own rule for each finished answer, before the visitor sees it (and before moderation).
   * Setting it holds every answer back until it is complete (docs/guardrails.md).
   */
  readonly onOutput?: GuardHook | undefined;
  /**
   * `untrusted` hands your tools' results to the model wrapped as third-party data that it must
   * not follow as instructions (prompt-injection hardening, ADR-0004). Recommended for tools that
   * return text from outside your control. Default `trusted`: results go to the model as they are.
   */
  readonly toolResults?: "trusted" | "untrusted" | undefined;
}

  readonly conversationsPerIpPerHour?: number;

  readonly messagesPerIpPerMinute?: number;

  /** File uploads per visitor per hour, when attachments are on. Default 20. */
  readonly uploadsPerVisitorPerHour?: number;
  /** File uploads per client IP per hour, across visitors. Default 60. */
  readonly uploadsPerIpPerHour?: number;

  readonly uploadsPerDay?: number;

  readonly transcriptionsPerVisitorPerHour?: number;

  readonly voiceSecondsPerDay?: number;
  /** Seconds of speech transcribed per client IP per UTC day. Default 600. */
  readonly voiceSecondsPerIpPerDay?: number;
}

/**
 * The retrieval port of ADR-0025: what a bot's own knowledge index is, with no OpenAI or database
 * types. Every model path and every adapter (in-memory, pgvector, community ones) speaks only these.
 * Server-side only: the client never chooses an index (ADR-0004).
 */
/**
 * One piece of a document: within one section, at most about 350 tokens. Core chunks every
 * document (`knowledgeDocument`), so every index stores the same chunks.
 */
interface KnowledgeChunk {
  /** What the model cites: `k` and the first 12 hex characters of `sha256`. Stable across syncs and turns. */
  readonly sourceId: string;
  /** The heading trail inside the document, such as "Pricing › Billing"; undefined before the first heading. */
  readonly section: string | undefined;
  readonly text: string;
  /** SHA-256 (hex) over the document title, the section and the text: the embedding cache key. */
  readonly sha256: string;
}
/** One file of the knowledge folder, extracted and chunked. Built by `askdare kb sync`, never at runtime. */
interface KnowledgeDocument {
  /** POSIX path relative to the knowledge folder: the document's identity in the index. */
  readonly path: string;
  /** SHA-256 (hex) of the file's bytes: an unchanged file is skipped. */
  readonly sha256: string;
  readonly title: string;
  /** http(s) only; the only link a citation may carry. */
  readonly url: string | undefined;
  /** In document order. */
  readonly chunks: readonly KnowledgeChunk[];
}
/** What one `sync` did. */
interface SyncReport {
  readonly added: readonly string[];
  readonly changed: readonly string[];
  readonly removed: readonly string[];
  /** Documents whose file hash, title and URL were already in the index. */
  readonly unchanged: number;
  /** Distinct chunks embedded in this run. */
  readonly embeddedChunks: number;
  /** Chunks of added or changed documents whose embedding the index already had. */
  readonly reusedChunks: number;
  readonly embeddingTokens: number;
}
/** A chunk with its document, as search and citations see it. `text` is untrusted third-party content. */
interface Passage {
  readonly sourceId: string;
  readonly path: string;
  readonly title: string;
  readonly url: string | undefined;
  readonly section: string | undefined;
  readonly text: string;
  /** Relevance from `search`, higher is better; absent from `passage()`. */
  readonly score?: number;
}
interface SearchOptions {
  readonly maxResults: number;
  readonly signal?: AbortSignal;
}
interface KnowledgeCallOptions {
  readonly signal?: AbortSignal;
}
/**
 * One finding of `diagnose`, for `askdare doctor`. Every index reports `knowledge.synced` and
 * `knowledge.embedder`; adapters add their own, such as `knowledge.pgvector`.
 */
interface KnowledgeCheck {
  readonly id: string;
  readonly ok: boolean;
  /** Counts, versions and what to do. Never document text. */
  readonly detail: string;
}
/** A bot's knowledge. One database may hold many bots' indexes, each under its own index ID. */
interface KnowledgeIndex {
  /**
   * Makes the index mirror `documents`, the whole folder. Idempotent: an unchanged file is skipped,
   * and only chunks whose hash the index doesn't have are embedded. Each document is written
   * atomically. Refuses documents that `checkKnowledgeDocuments` refuses, before changing anything.
   */
  sync(documents: readonly KnowledgeDocument[], options?: KnowledgeCallOptions): Promise<SyncReport>;
  /** The best passages first, each with a score. */
  search(query: string, options: SearchOptions): Promise<readonly Passage[]>;
  /** Resolves a cited source ID from this turn or an earlier one; undefined once the chunk is gone. */
  passage(sourceId: string, options?: KnowledgeCallOptions): Promise<Passage | undefined>;
  /** Removes everything this index holds. */
  clear(options?: KnowledgeCallOptions): Promise<void>;
  diagnose(options?: KnowledgeCallOptions): Promise<readonly KnowledgeCheck[]>;
}
interface Embeddings {
  /** One vector per input, in input order, each `dimensions` long. */
  readonly vectors: readonly (readonly number[])[];
  readonly tokens: number;
}
/** Turns text into vectors. Separate from the index, so the site picks the key and the model. */
interface Embedder {
  /**
   * Provider, model and dimensions, such as `openai:<model>:1536`. Stored with the index: a
   * different ID makes the next sync embed everything again.
   */
  readonly id: string;
  readonly dimensions: number;
  embed(texts: readonly string[], options?: KnowledgeCallOptions): Promise<Embeddings>;
}
/**
 * The `ConversationStore` port (ADR-0024): where conversations live. `openaiConversations()` keeps
 * them at OpenAI (the default), `memoryConversations()` in memory for tests, and add-ons such as
 * `@boldare/askdare-postgres` in a database. Nothing here depends on a model provider: the mapping
 * between stored items and a provider's items lives with that provider (ADR-0002).
 */
/** Who may resume, read and delete a conversation (ADR-0009). */
interface ConversationOwner {
  readonly botId: string;
  /**
   * `visitor:<id>` for an anonymous visitor, `user:<hmac>` for a signed-in user. Opaque to the
   * store: it stores and returns the string, and the server compares it.
   */
  readonly owner: string;
}
/** A stored conversation, as `info()` describes it. */
interface ConversationInfo extends ConversationOwner {
  readonly id: string;
  /** Epoch milliseconds. */
  readonly createdAt: number;
  /** Epoch milliseconds of the last append; retention counts from here. */
  readonly updatedAt: number;
  /** Sequence number of the last item, 0 while empty. `append`'s `expect` compares with it. */
  readonly version: number;
}
/** Where a cited passage came from, stored with the text that cites it (ADR-0024, decision 8). */
interface StoredCitation {
  readonly title: string;
  readonly url?: string | undefined;
  /** The knowledge source's ID, when the passage came from the bot's knowledge base. */
  readonly sourceId?: string | undefined;
}
/** One part of a stored message. */
type StoredPart = {
  readonly type: "text";
  readonly text: string;
  readonly citations?: readonly StoredCitation[] | undefined;
} | {
  readonly type: "page_context";
  readonly text: string;
} | {
  readonly type: "file";
  readonly fileId: string;
  readonly kind: "image" | "pdf";
  readonly name?: string | undefined;
} | {
  readonly type: "refusal";
  readonly text: string;
};
/**
 * One conversation item. Its `id` is chosen by the writer and unique within the conversation; a
 * store that assigns its own IDs (OpenAI) may return others.
 */
type StoredItem = {
  readonly id: string;
  readonly type: "message";
  readonly role: "user" | "assistant" | "developer";
  /** Assistant text written before a tool call is `commentary`; the answer is `final_answer`. */
  readonly phase?: "commentary" | "final_answer" | undefined;
  readonly parts: readonly StoredPart[];
} | {
  readonly id: string;
  readonly type: "call";
  readonly callId: string;
  readonly name: string;
  /** JSON, as the model wrote it. */
  readonly arguments: string;
} | {
  readonly id: string;
  readonly type: "call_output";
  readonly callId: string;
  readonly output: string;
} | {
  readonly id: string;
  readonly type: "native";
  readonly provider: string;
  readonly family?: string | undefined;
  readonly data: unknown;
};
/** A stored item with its position in the conversation. */
type PositionedItem = StoredItem & {
  /**
   * The item's position: 1 for the first item of the conversation, one more per item. A provider
   * store that can't number its items (OpenAI) counts within the page only.
   */
  readonly seq: number;
  /** Epoch milliseconds of the append that wrote it; 0 when the store doesn't know. */
  readonly at: number;
};
/** What the next model call reads to continue a conversation. */
type ConversationContext = {
  readonly kind: "provider";
  readonly provider: string;
  readonly ref: string;
  readonly version: number;
} | {
  readonly kind: "items";
  readonly items: readonly PositionedItem[];
  readonly version: number;
};
/** One page of a conversation's items. */
interface ConversationPage {
  /** Oldest first. */
  readonly items: readonly PositionedItem[];
  /** Pass as `before` for the next older page; absent at the start of the conversation. */
  readonly before?: string | undefined;
}
/** One finding of a store's `diagnose()`, shown by `askdare doctor`. Never contains secrets. */
interface ConversationStoreCheck {
  readonly status: "ok" | "warn" | "fail";
  readonly title: string;
  /** What to do about a warning or failure. */
  readonly fix?: string | undefined;
}
/**
 * Thrown by `append` when the conversation is no longer at the `expect`ed version: another turn
 * wrote first. Nothing of the refused append is stored.
 */
declare class VersionConflictError extends Error {
  readonly expected: number;
  readonly actual: number;
  constructor(expected: number, actual: number);
}
/**
 * Where conversations live (ADR-0024). Every method that takes an ID answers an unknown or deleted
 * conversation the same way: `info` with `undefined`, `delete` with no files, the others by
 * throwing. Implementations never log stored content.
 */
interface ConversationStore {
  /** Stable name for logs and diagnostics, such as `openai-conversations` or `postgres`. */
  readonly name: string;
  /** Creates an empty conversation and returns its ID. */
  create(owner: ConversationOwner, signal?: AbortSignal): Promise<string>;
  /** The conversation's bot and owner, which the server compares with the resume token's. */
  info(id: string, signal?: AbortSignal): Promise<ConversationInfo | undefined>;
  /**
   * What the next model call reads. An item store returns the latest `maxItems` items at most,
   * cut forward to a visitor message so a turn is never split.
   */
  context(id: string, options?: {
    readonly maxItems?: number | undefined;
    readonly signal?: AbortSignal | undefined;
  }): Promise<ConversationContext>;
  /**
   * Appends items in order, all or none. With `expect`, throws {@link VersionConflictError} unless
   * the conversation is still at that version. Resolves to the new version. An item ID already in
   * the conversation is refused.
   */
  append(id: string, items: readonly StoredItem[], options?: {
    readonly expect?: number | undefined;
    readonly signal?: AbortSignal | undefined;
  }): Promise<number>;
  /** The newest items before `before` (all when absent), oldest first. `before` is opaque. */
  page(id: string, options?: {
    readonly before?: string | undefined;
    readonly limit?: number | undefined;
    readonly signal?: AbortSignal | undefined;
  }): Promise<ConversationPage>;
  /**
   * Deletes the conversation and returns the IDs of the files its visitor attached, for the
   * server to delete. Deleting an unknown conversation is not an error.
   */
  delete(id: string, signal?: AbortSignal): Promise<{
    readonly fileIds: readonly string[];
  }>;
  /**
   * Retention: deletes up to `limit` conversations with no append since `idleSince` (epoch
   * milliseconds) and returns them, so the caller deletes their files too. Absent on stores that
   * can't list their conversations (OpenAI).
   */
  sweep?(idleSince: number, limit: number): Promise<readonly {
    readonly id: string;
    readonly fileIds: readonly string[];
  }[]>;
  /** Small live checks for `askdare doctor`, such as a reachable database with the right schema. */
  diagnose?(signal?: AbortSignal): Promise<readonly ConversationStoreCheck[]>;
}
/** Result of incrementing a fixed-window counter. */
interface CounterHit {
  /** Counter value after the increment. */
  readonly count: number;
  /** Milliseconds until the window resets. */
  readonly resetMs: number;
}
/**
 * Shared counter storage behind rate limits and budgets. The in-memory default works for a single
 * instance; multi-instance deployments (Vercel, several dynos) need a shared store such as
 * `@boldare/askdare-upstash`. Implementations must be atomic per key.
 */
interface RateLimitStore {
  /**
   * Adds `amount` to `key`, starting a `windowMs` window if none is open. `0` reads without
   * changing; negative amounts refund.
   */
  hit(key: string, windowMs: number, amount: number): Promise<CounterHit>;
  /** True for stores that are shared across server instances. */
  readonly shared: boolean;
}

/**

  toBe(expected: unknown): void;
  toEqual(expected: unknown): void;

  toBeGreaterThan(expected: number): void;
};
